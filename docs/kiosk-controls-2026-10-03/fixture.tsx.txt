import {useState} from 'react'; import{createRoot}from'react-dom/client';
import{QueryClient,QueryClientProvider}from'@tanstack/react-query';
import './src/index.css';import{RadialDial}from'./src/components/glass/RadialDial';import{DragSlider}from'./src/components/glass/DragSlider';import{ClimateDetail}from'./src/components/contextual/ClimateDetail';
import{useEntityStore}from'./src/store/entities';import{useHAEntity}from'./src/hooks/useHAEntity';import type{HassEntity}from'home-assistant-js-websocket';
const ent={entity_id:'climate.test',state:'heat',attributes:{temperature:20.25,current_temperature:19.5,min_temp:7.25,max_temp:35.25,target_temp_step:.25,hvac_modes:['off','heat','cool'],temperature_unit:'°C'},last_changed:'',last_updated:'',context:{id:'fixture',parent_id:null,user_id:null}} satisfies HassEntity;
useEntityStore.getState().setEntities({'climate.test':ent});useEntityStore.getState().setConnected(true);
const q=new QueryClient({defaultOptions:{queries:{retry:false}}});
function App(){const[v,setV]=useState(20.25);const[commits,setCommits]=useState<number[]>([]);const[vol,setVol]=useState(25);const[e,setE]=useState(false);const entity=useHAEntity('climate.test')!;
return <main style={{padding:16,background:'var(--canvas-page)',color:'var(--ink)',minHeight:'100dvh'}}><h1>Laboratorio controlli · nessun dispositivo reale</h1><button onClick={()=>document.documentElement.classList.toggle('dark')}>Cambia tema</button><button onClick={()=>setV(24.25)}>Aggiornamento remoto 24,25</button><button onClick={()=>{setE(!e);useEntityStore.getState().setEntities({'climate.test':{...ent,attributes:{...ent.attributes,temperature:!e?null:20.25}}})}}>Target assente</button><div style={{maxWidth:340,margin:'auto'}}><RadialDial value={v} min={7.25} max={35.25} step={.25} label={String(v)} ariaLabel="Ruota di prova" size={236} onChange={setV} onCommit={n=>setCommits(a=>[...a,n])}/><output aria-label="Comandi ruota">{commits.join(', ')||'Nessuno'}</output><DragSlider value={vol} onChange={setVol} onChangeEnd={n=>setCommits(a=>[...a,n])} ariaLabel="Slider di prova"/><ClimateDetail entity={entity}/></div></main>}
createRoot(document.getElementById('root')!).render(<QueryClientProvider client={q}><App/></QueryClientProvider>);
